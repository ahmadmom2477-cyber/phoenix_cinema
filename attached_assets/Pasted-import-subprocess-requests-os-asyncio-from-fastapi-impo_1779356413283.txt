import subprocess, requests, os, asyncio
from fastapi import FastAPI
from fastapi.responses import HTMLResponse
import uvicorn

app = FastAPI()
OMDB_API_KEY = "58e58c4f"

# دالة تنظيف العمليات السحابية (لضمان استقرار Render)
def cleanup():
    os.system("pkill -f peerflix")

@app.get("/", response_class=HTMLResponse)
def index():
    return """
    <html>
        <body style="background:#000; color:#fff; text-align:center; padding:20px; font-family:sans-serif;">
            <h1>🎬 AM Cinema v5 - Final Build</h1>
            <input id="n" placeholder="اسم الفيلم أو المسلسل" style="padding:10px; width:90%; border-radius:5px;"><br><br>
            <input id="s" placeholder="موسم (للمسلسلات)" style="padding:10px; width:40%">
            <input id="e" placeholder="حلقة (للمسلسلات)" style="padding:10px; width:40%"><br><br>
            <button onclick="run()" style="padding:15px; width:90%; background:#e50914; color:white; border:none; border-radius:5px;">تشغيل</button>
            <div id="d" style="margin-top:20px;"></div>
            <script>
                async function run() {
                    const n = document.getElementById('n').value;
                    const s = document.getElementById('s').value;
                    const e = document.getElementById('e').value;
                    document.getElementById('d').innerText = "⏳ جاري المعالجة...";
                    const res = await fetch(`/play?n=${encodeURIComponent(n)}&s=${s}&e=${e}`);
                    const j = await res.json();
                    document.getElementById('d').innerHTML = j.url ? 
                        `✅ الرابط جاهز:<br><code style="background:#333; padding:10px;">${j.url}</code>` : `❌ ${j.err}`;
                }
            </script>
        </body>
    </html>
    """

@app.get("/play")
async def play(n: str, s: str = "", e: str = ""):
    cleanup()
    
    # 1. حالة المسلسل (EZTV)
    if s and e:
        slug = n.lower().replace(" ", "-")
        # رابط بحث افتراضي للمسلسل
        torrent_url = f"https://eztv.re/search/{slug}" 
        # ملاحظة: منطق استخراج رابط الـ Magnet/Torrent من صفحة المسلسل يحتاج دقة في التحليل
    
    # 2. حالة الفيلم (YTS)
    else:
        info = requests.get(f"https://www.omdbapi.com/?t={n}&apikey={OMDB_API_KEY}").json()
        if info.get("Response") != "True": return {"err": "لم يتم العثور على الفيلم"}
        
        slug = f"{info['Title'].lower().replace(' ', '-')}-{info['Year']}"
        res = requests.get(f"https://yts.vg/movie/{slug}", headers={"User-Agent": "Mozilla/5.0"})
        import re
        match = re.search(r'/movie/(\d+)/download', res.text)
        if not match: return {"err": "غير متاح للتحميل"}
        torrent_url = f"https://yts.vg/movie/{match.group(1)}/download/1080p/bluray/x264"

    # تشغيل المشغل
    subprocess.Popen(f'peerflix "{torrent_url}" --port 8888 --address 0.0.0.0 --path /tmp --quiet &', shell=True)
    return {"url": "http://YOUR_RENDER_URL:8888"}

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)
